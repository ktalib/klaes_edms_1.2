🔍 Step Navigation Debug Console

Open your actual Primary Form page and paste this into the browser console (F12):
// === STEP NAVIGATION DIAGNOSTIC ===
console.log('🔍 === STEP NAVIGATION DIAGNOSTIC ===');

// 1. Check all form sections
const allSteps = document.querySelectorAll('.form-section');
console.log(`📋 Total form-section elements found: ${allSteps.length}`);

allSteps.forEach((step, index) => {
    const computed = window.getComputedStyle(step);
    console.log(`
    Step ${index + 1} (${step.id}):
    - Classes: ${step.className}
    - Display (CSS): ${computed.display}
    - Display (inline): ${step.style.display}
    - Visibility: ${computed.visibility}
    - Opacity: ${computed.opacity}
    - Height: ${computed.height}
    - Has 'active' class: ${step.classList.contains('active')}
    - IS VISIBLE: ${computed.display !== 'none' && computed.visibility !== 'hidden' && computed.opacity !== '0'}
    `);
});

// 2. Check if goToStep function exists
console.log(`\n🎯 goToStep function exists: ${typeof window.goToStep === 'function'}`);
if (typeof window.goToStep === 'function') {
    console.log('   Function source:', window.goToStep.toString().substring(0, 200));
}

// 3. Check CSS rules affecting .form-section
console.log('\n📐 CSS Rules for .form-section:');
const testDiv = document.createElement('div');
testDiv.className = 'form-section';
document.body.appendChild(testDiv);
const rules = window.getComputedStyle(testDiv);
console.log('   display:', rules.display);
console.log('   visibility:', rules.visibility);
console.log('   opacity:', rules.opacity);
document.body.removeChild(testDiv);

// 4. Check .form-section.active
console.log('\n📐 CSS Rules for .form-section.active:');
const testDiv2 = document.createElement('div');
testDiv2.className = 'form-section active';
document.body.appendChild(testDiv2);
const rules2 = window.getComputedStyle(testDiv2);
console.log('   display:', rules2.display);
console.log('   visibility:', rules2.visibility);
console.log('   opacity:', rules2.opacity);
document.body.removeChild(testDiv2);

// 5. Try to navigate to step 2
console.log('\n🚀 Attempting to navigate to Step 2...');
if (typeof window.goToStep === 'function') {
    window.goToStep(2);
    
    setTimeout(() => {
        console.log('\n✅ After navigation check:');
        document.querySelectorAll('.form-section').forEach(step => {
            const computed = window.getComputedStyle(step);
            if (computed.display !== 'none') {
                console.log(`   VISIBLE: ${step.id} (${step.className})`);
            }
        });
    }, 500);
} else {
    console.error('❌ goToStep function not found!');
}

console.log('\n=== END DIAGNOSTIC ===');
